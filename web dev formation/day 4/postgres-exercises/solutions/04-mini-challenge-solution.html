Solution 04 — Mini Challenge: Online Store

Complete SQL solution for the products table, queries, and updates.

Step 1 Create the products Table

CREATE TABLE products (
    id       SERIAL PRIMARY KEY,
    name     TEXT NOT NULL,
    category TEXT NOT NULL,
    price    NUMERIC(8, 2) NOT NULL,
    stock    INT DEFAULT 0
);

Step 2 Insert 6 Products

INSERT INTO products (name, category, price, stock) VALUES
('Wireless Mouse',      'Electronics', 25.99,  40),
('Mechanical Keyboard', 'Electronics', 89.50,  15),
('Coffee Mug',          'Kitchen',     12.00,   0),
('Standing Desk',       'Furniture',  320.00,   5),
('USB-C Cable',         'Electronics',  9.99, 100),
('Desk Lamp',           'Furniture',   45.00,   0);

Step 3 Query & Filter the Store

-- 1. Electronics in stock
SELECT *
FROM products
WHERE category = 'Electronics'
  AND stock > 0;

-- 2. Out of stock products
SELECT *
FROM products
WHERE stock = 0;

-- 3. 2 cheapest products
SELECT *
FROM products
ORDER BY price ASC
LIMIT 2;

-- 4. Name contains 'Desk' and price < 100
SELECT *
FROM products
WHERE name ILIKE '%desk%'
  AND price < 100;

Step 4 Bonus: Safe UPDATE & DELETE

-- 1. Restock Coffee Mug
UPDATE products
SET stock = 25
WHERE name = 'Coffee Mug';

-- 2. Delete Desk Lamp
DELETE FROM products
WHERE name = 'Desk Lamp';
