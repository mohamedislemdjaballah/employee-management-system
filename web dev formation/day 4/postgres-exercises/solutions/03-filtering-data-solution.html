Solution 03 — Filtering Data with WHERE

Reference SQL queries for comparisons, logical operators, ranges, patterns, and NULL checks.

Part A — Basic Comparisons

1. Employees in Engineering

SELECT *
FROM employees
WHERE department = 'Engineering';

2. Salary strictly greater than 70000

SELECT *
FROM employees
WHERE salary > 70000;

3. Not in Marketing

SELECT *
FROM employees
WHERE department != 'Marketing';
Part B — Combining Conditions & Ranges

4. Engineering earning at least 80000

SELECT *
FROM employees
WHERE department = 'Engineering'
  AND salary >= 80000;

5. In HR or Sales

SELECT *
FROM employees
WHERE department IN ('HR', 'Sales');

-- Alternative using OR:
-- WHERE department = 'HR' OR department = 'Sales';

6. Hired between 2020 and 2022

SELECT *
FROM employees
WHERE hire_year BETWEEN 2020 AND 2022;

7. Salary between 55000 and 65000, sorted descending

SELECT *
FROM employees
WHERE salary BETWEEN 55000 AND 65000
ORDER BY salary DESC;
Part C — Patterns & NULL Values

8. First name starts with 'A' or 'a'

SELECT *
FROM employees
WHERE first_name ILIKE 'a%';

9. Last name contains 'o'

SELECT *
FROM employees
WHERE last_name ILIKE '%o%';

10. Missing email address

SELECT *
FROM employees
WHERE email IS NULL;

11. Has email and works in Engineering

SELECT *
FROM employees
WHERE email IS NOT NULL
  AND department = 'Engineering';
